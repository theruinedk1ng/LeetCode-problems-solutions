693. Binary Number with Alternating Bits

```js
/**
 * @param {number} n
 * @return {boolean}
 */
var hasAlternatingBits = function(n) {
    let bin = n.toString(2), previous = bin[0];
    const len = bin.length;

    for (let i = 1; i < len; i ++) {
        if (bin[i] === previous)
            return false;
        previous = bin[i];
    };

    return true;
};
```
